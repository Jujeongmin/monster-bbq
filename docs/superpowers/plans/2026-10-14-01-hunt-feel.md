```ts
// A point on the hunting ground in world pixels: x to the right, y down (bigger y = nearer the viewer).
export interface Vec {
  x: number;
  y: number;
}

// The walkable band of the ground. y is where the feet are.
export interface Field {
  minX: number;
  maxX: number;
  minY: number;
  maxY: number;
}
```

`shared/hunt/geometry.ts`:

```ts
import type { Field, Vec } from "./types";

export function clampToField(p: Vec, f: Field): Vec {
  return {
    x: Math.min(f.maxX, Math.max(f.minX, p.x)),
    y: Math.min(f.maxY, Math.max(f.minY, p.y)),
  };
}

// Belt-scroller depth: two things can only hit each other when they stand on (nearly) the same line.
export function sameLane(a: Vec, b: Vec, tolerance: number): boolean {
  return Math.abs(a.y - b.y) <= tolerance;
}

export function distance(a: Vec, b: Vec): number {
  return Math.hypot(a.x - b.x, a.y - b.y);
}

export function stepToward(from: Vec, to: Vec, maxDist: number): { pos: Vec; arrived: boolean } {
  const d = distance(from, to);
  if (d <= maxDist) return { pos: { x: to.x, y: to.y }, arrived: true };
  const k = maxDist / d;
  return { pos: { x: from.x + (to.x - from.x) * k, y: from.y + (to.y - from.y) * k }, arrived: false };
}

// The one drawn in front: lowest on screen.
export function frontmost<T extends { pos: Vec }>(items: T[]): T | undefined {
  return items.reduce<T | undefined>((best, it) => (!best || it.pos.y > best.pos.y ? it : best), undefined);
}
```

`shared/hunt/combat.ts`:

```ts
import { sameLane } from "./geometry";
import type { Vec } from "./types";

export type WeaponKind = "melee" | "ranged";

export interface WeaponStats {
  kind: WeaponKind;
  range: number;
  damage: number;
  cooldownMs: number;
  // how long the attack animation locks the hunter in place
  swingMs: number;
}

export function inAttackRange(attacker: Vec, target: Vec, weapon: WeaponStats, laneTolerance: number): boolean {
  return sameLane(attacker, target, laneTolerance) && Math.abs(attacker.x - target.x) <= weapon.range;
}

// Where to stand to hit `target`: on its lane, a little inside reach, on the side the attacker is on.
// Already close enough on x means only the lane changes.
export function approachPoint(attacker: Vec, target: Vec, weapon: WeaponStats): Vec {
  const dx = attacker.x - target.x;
  const reach = weapon.range * 0.8;
  const x = Math.abs(dx) <= reach ? attacker.x : target.x + Math.sign(dx || -1) * reach;
  return { x, y: target.y };
}

export type Order = { kind: "none" } | { kind: "moveTo"; to: Vec } | { kind: "attack"; target: Vec };
export type PlayerAction = { kind: "idle" } | { kind: "move"; to: Vec } | { kind: "attack" };

export function decidePlayerAction(pos: Vec, order: Order, weapon: WeaponStats, laneTolerance: number): PlayerAction {
  switch (order.kind) {
    case "none":
      return { kind: "idle" };
    case "moveTo":
      return pos.x === order.to.x && pos.y === order.to.y ? { kind: "idle" } : { kind: "move", to: order.to };
    case "attack":
      return inAttackRange(pos, order.target, weapon, laneTolerance)
        ? { kind: "attack" }
        : { kind: "move", to: approachPoint(pos, order.target, weapon) };
  }
}

export interface HitTuning {
  comboSteps: number;
  comboWindowMs: number;
  finisherMult: number;
  counterMult: number;
  backAttackMult: number;
  knockback: number;
  finisherKnockback: number;
}

// Melee swings on the same target inside the window chain 1 -> 2 -> ... -> comboSteps, then start over.
export function nextComboStep(prevStep: number, msSinceLastSwing: number, sameTarget: boolean, weapon: WeaponKind, tuning: HitTuning): number {
  if (weapon !== "melee" || !sameTarget || msSinceLastSwing > tuning.comboWindowMs || prevStep >= tuning.comboSteps) return 1;
  return prevStep + 1;
}

export interface HitInput {
  weapon: WeaponStats;
  comboStep: number;
  // the target is open (a charger catching its breath after a charge): counter bonus
  targetVulnerable: boolean;
  attackerX: number;
  targetX: number;
  targetFacing: 1 | -1;
}

export interface HitResult {
  damage: number;
  finisher: boolean;
  counter: boolean;
  backAttack: boolean;
  knockback: number;
}

export function computeHit(input: HitInput, tuning: HitTuning): HitResult {
  const finisher = input.weapon.kind === "melee" && input.comboStep >= tuning.comboSteps;
  const counter = input.targetVulnerable;
  // behind = standing on the side opposite to where the target faces
  const backAttack = (input.attackerX - input.targetX) * input.targetFacing < 0;
  let mult = 1;
  if (finisher) mult *= tuning.finisherMult;
  if (counter) mult *= tuning.counterMult;
  if (backAttack) mult *= tuning.backAttackMult;
  return {
    damage: Math.round(input.weapon.damage * mult),
    finisher,
    counter,
    backAttack,
    knockback: finisher ? tuning.finisherKnockback : tuning.knockback,
  };
}

export interface HitLog {
  meleeHits: number;
  rangedHits: number;
  lastHit: WeaponKind | null;
}

export const EMPTY_HIT_LOG: HitLog = { meleeHits: 0, rangedHits: 0, lastHit: null };

export function recordHit(log: HitLog, kind: WeaponKind): HitLog {
  return kind === "melee"
    ? { ...log, meleeHits: log.meleeHits + 1, lastHit: kind }
    : { ...log, rangedHits: log.rangedHits + 1, lastHit: kind };
}

export type MeatGrade = "premium" | "normal";

// Spec 3: premium when the killing blow was melee and it took at most `maxRangedHits` ranged hits.
export function meatGrade(log: HitLog, maxRangedHits: number): MeatGrade {
  return log.lastHit === "melee" && log.rangedHits <= maxRangedHits ? "premium" : "normal";
}
```

```ts
export type ChargerPhase = "wander" | "telegraph" | "charge" | "recover";

export interface ChargerState {
  pos: Vec;
  phase: ChargerPhase;
  phaseMs: number;
  facing: 1 | -1;
  wanderTarget: Vec | null;
  hitThisCharge: boolean;
}

export function newCharger(pos: Vec): ChargerState {
  return { pos, phase: "wander", phaseMs: 0, facing: -1, wanderTarget: null, hitThisCharge: false };
}

// Catching its breath after a charge: hits on it now count as counters.
export function isVulnerable(s: ChargerState): boolean {
  return s.phase === "recover";
}

export function updateCharger(
  s: ChargerState,
  player: Vec,
  dtMs: number,
  t: ChargerTuning,
  field: Field,
  rand: () => number,
): { state: ChargerState; hit: boolean } {
  const dt = dtMs / 1000;
  const phaseMs = s.phaseMs + dtMs;
  switch (s.phase) {
    case "wander": {
      const dx = player.x - s.pos.x;
      if (Math.abs(dx) <= t.aggroRange) {
        const facing: 1 | -1 = dx >= 0 ? 1 : -1;
        if (sameLane(s.pos, player, t.laneTolerance)) {
          return { state: { ...s, phase: "telegraph", phaseMs: 0, facing, wanderTarget: null, hitThisCharge: false }, hit: false };
        }
        const step = stepToward(s.pos, { x: s.pos.x, y: player.y }, t.wanderSpeed * dt);
        return { state: { ...s, pos: clampToField(step.pos, field), phaseMs, facing }, hit: false };
      }
      const target = s.wanderTarget ?? {
        x: field.minX + rand() * (field.maxX - field.minX),
        y: field.minY + rand() * (field.maxY - field.minY),
      };
      const step = stepToward(s.pos, target, t.wanderSpeed * dt);
      const facing: 1 | -1 = target.x >= s.pos.x ? 1 : -1;
      return {
        state: { ...s, pos: clampToField(step.pos, field), phaseMs, facing, wanderTarget: step.arrived ? null : target },
        hit: false,
      };
    }
    case "telegraph":
      return {
        state: phaseMs >= t.telegraphMs ? { ...s, phase: "charge", phaseMs: 0, hitThisCharge: false } : { ...s, phaseMs },
        hit: false,
      };
    case "charge": {
      const pos = clampToField({ x: s.pos.x + s.facing * t.chargeSpeed * dt, y: s.pos.y }, field);
      const hit = !s.hitThisCharge && sameLane(pos, player, t.laneTolerance) && Math.abs(pos.x - player.x) <= t.hitWidth;
      const done = phaseMs >= t.chargeMs;
      return {
        state: { ...s, pos, phase: done ? "recover" : "charge", phaseMs: done ? 0 : phaseMs, hitThisCharge: s.hitThisCharge || hit },
        hit,
      };
    }
    case "recover":
      return { state: phaseMs >= t.recoverMs ? { ...s, phase: "wander", phaseMs: 0 } : { ...s, phaseMs }, hit: false };
  }
}
```

```ts
  private onTap(p: Phaser.Input.Pointer, over: Phaser.GameObjects.GameObject[]) {
    if (this.player.down || over.includes(this.weaponButton)) return;
    const tapped = this.boars.filter((b) => !b.dead && over.includes(b.sprite)).map((b) => ({ pos: b.ai.pos, boar: b }));
    const target = frontmost(tapped)?.boar;
    if (target) {
      this.targetId = target.id;
      this.order = { kind: "attack", target: target.ai.pos };
    } else {
      this.targetId = null;
      this.order = { kind: "moveTo", to: clampToField({ x: p.worldX, y: p.worldY }, H.field) };
    }
  }

  private toggleWeapon() {
    if (this.player.down) return;
    this.player.weapon = this.player.weapon === "melee" ? "ranged" : "melee";
    this.player.comboStep = 0;
    this.refreshWeaponLabel();
  }

  private refreshWeaponLabel() {
    this.weaponButton.setText(this.player.weapon === "melee" ? "근접 ⇄ 원거리" : "원거리 ⇄ 근접");
  }

  private updatePlayer(dt: number) {
    const p = this.player;
    p.cooldownMs = Math.max(0, p.cooldownMs - dt);
    p.swingMs = Math.max(0, p.swingMs - dt);
    p.msSinceSwing += dt;

    // keep an attack order pointed at the boar's current position; drop it once the boar is gone
    if (this.order.kind === "attack") {
      const target = this.liveBoar(this.targetId);
      this.order = target ? { kind: "attack", target: target.ai.pos } : { kind: "none" };
    }
    // a swing always finishes before the next order is followed
    if (p.swingMs > 0) return;

    const weapon = H.weapons[p.weapon];
    const action = decidePlayerAction(p.pos, this.order, weapon, H.laneTolerance);
    if (action.kind === "move") {
      const step = stepToward(p.pos, action.to, (H.player.speed * dt) / 1000);
      if (step.pos.x !== p.pos.x) this.hunter.setFlipX(step.pos.x < p.pos.x);
      p.pos = step.pos;
      if (step.arrived && this.order.kind === "moveTo") this.order = { kind: "none" };
      this.loop("hunter_walk");
    } else if (action.kind === "attack" && this.targetId !== null) {
      const target = this.liveBoar(this.targetId)!;
      this.hunter.setFlipX(target.ai.pos.x < p.pos.x);
      if (p.cooldownMs === 0) this.swing(target);
      else this.loop("hunter_idle");
    } else {
      this.loop("hunter_idle");
    }
    this.hunter.setPosition(p.pos.x, p.pos.y).setDepth(p.pos.y);
  }

  private swing(target: Boar) {
    const p = this.player;
    const weapon = H.weapons[p.weapon];
    p.comboStep = nextComboStep(p.comboStep, p.msSinceSwing, p.lastTargetId === target.id, p.weapon, H.hit);
    p.lastTargetId = target.id;
    p.msSinceSwing = 0;
    p.cooldownMs = weapon.cooldownMs;
    p.swingMs = weapon.swingMs;
    this.hunter.play(p.weapon === "melee" ? "hunter_attack" : "hunter_throw");
    const kind = p.weapon;
    const comboStep = p.comboStep;
    // the blow lands mid-swing
    this.time.delayedCall(weapon.swingMs * 0.5, () => this.landHit(target.id, kind, comboStep));
  }

  private landHit(boarId: number, kind: WeaponKind, comboStep: number) {
    const b = this.liveBoar(boarId);
    const p = this.player;
    if (!b || p.down) return;
    // the boar may have run out of reach during the swing: a whiff
    if (!inAttackRange(p.pos, b.ai.pos, H.weapons[kind], H.laneTolerance)) return;

    const hit = computeHit(
      { weapon: H.weapons[kind], comboStep, targetVulnerable: isVulnerable(b.ai), attackerX: p.pos.x, targetX: b.ai.pos.x, targetFacing: b.ai.facing },
      H.hit,
    );
    b.hp -= hit.damage;
    b.hits = recordHit(b.hits, kind);
    const away = b.ai.pos.x >= p.pos.x ? 1 : -1;
    b.ai = { ...b.ai, pos: clampToField({ x: b.ai.pos.x + away * hit.knockback, y: b.ai.pos.y }, H.field) };

    // game feel: freeze, flash, shake on big hits, pop the number and the tags
    this.hitStop(hit.finisher || hit.counter ? H.hitStopMs * 2 : H.hitStopMs);
    b.sprite.setTintFill(0xffffff);
    this.time.delayedCall(70, () => b.sprite.active && b.sprite.clearTint());
    if (hit.finisher || hit.counter) this.cameras.main.shake(120, 0.012);
    const tags = [hit.counter && "카운터!", hit.backAttack && "백어택!", hit.finisher && "막타!"].filter(Boolean).join(" ");
    this.popText(b.ai.pos.x, b.ai.pos.y - 52, `${hit.damage}${tags ? " " + tags : ""}`, hit.counter || hit.finisher ? "#ffd166" : "#ffffff");

    if (b.hp <= 0) this.killBoar(b);
  }

  private killBoar(b: Boar) {
    b.dead = true;
    b.sprite.disableInteractive();
    if (this.targetId === b.id) {
      this.targetId = null;
      this.order = { kind: "none" };
    }
    const premium = meatGrade(b.hits, H.premiumMaxRangedHits) === "premium";
    this.popText(b.ai.pos.x, b.ai.pos.y - 70, premium ? "삼겹살 ★특급" : "삼겹살 (보통)", premium ? "#ffd166" : "#dddddd", 1400);
    this.tweens.add({
      targets: b.sprite,
      alpha: 0,
      y: b.sprite.y - 6,
      duration: 500,
      onComplete: () => {
        b.sprite.destroy();
        this.boars = this.boars.filter((x) => x !== b);
        this.time.delayedCall(2000, () => this.spawnBoar());
      },
    });
  }

  private updateBoar(b: Boar, dt: number) {
    if (b.dead) return;
    const before = b.ai.phase;
    // a knocked-out hunter is ignored: send the boars wandering
    const seen = this.player.down ? { x: -99999, y: -99999 } : this.player.pos;
    const r = updateCharger(b.ai, seen, dt, H.boar, H.field, Math.random);
    b.ai = r.state;
    if (r.hit) this.hurtPlayer(H.boar.contactDamage, b.ai.facing);
    b.sprite.setPosition(b.ai.pos.x, b.ai.pos.y).setDepth(b.ai.pos.y).setFlipX(b.ai.facing < 0);
    if (b.ai.phase !== before) this.showPhase(b);
  }

  private showPhase(b: Boar) {
    const s = b.sprite;
    this.tweens.killTweensOf(s);
    s.setAlpha(1).clearTint();
    switch (b.ai.phase) {
      case "telegraph":
        s.anims.stop();
        s.setTexture("boar_attack_0").setTint(0xff6060);
        this.tweens.add({ targets: s, alpha: 0.45, yoyo: true, repeat: 3, duration: H.boar.telegraphMs / 8 });
        this.popText(b.ai.pos.x, b.ai.pos.y - 60, "!", "#ff5050", H.boar.telegraphMs);
        break;
      case "charge":
        s.play("boar_charge");
        break;
      case "recover":
        s.anims.stop();
        s.setTexture("boar_idle");
        break;
      case "wander":
        s.play("boar_walk");
        break;
    }
  }

  private hurtPlayer(damage: number, dir: 1 | -1) {
    const p = this.player;
    if (p.down) return;
    p.hp = Math.max(0, p.hp - damage);
    p.pos = clampToField({ x: p.pos.x + dir * 24, y: p.pos.y }, H.field);
    p.swingMs = 0;
    this.hunter.setPosition(p.pos.x, p.pos.y).setTint(0xff8080);
    this.time.delayedCall(150, () => this.hunter.clearTint());
    this.cameras.main.shake(100, 0.01);
    this.popText(p.pos.x, p.pos.y - 50, `-${damage}`, "#ff6060");
    if (p.hp === 0) {
      p.down = true;
      this.order = { kind: "none" };
      this.hunter.anims.stop();
      this.hunter.setTint(0x777777).setAngle(-90);
      this.add
        .text(GAME_WIDTH / 2, GAME_HEIGHT / 2, "쓰러졌다… 2초 뒤 다시", { fontFamily: "monospace", fontSize: "12px", color: "#ffffff", backgroundColor: "#000000aa", padding: { x: 6, y: 3 } })
        .setOrigin(0.5)
        .setScrollFactor(0)
        .setDepth(1001);
      this.time.delayedCall(2000, () => this.scene.restart());
    }
  }

  private spawnBoar() {
    const f = H.field;
    let x = f.minX + Math.random() * (f.maxX - f.minX);
    // not right on top of the hunter
    if (Math.abs(x - this.player.pos.x) < 160) x = Math.min(f.maxX, Math.max(f.minX, x + 220 * (x < this.player.pos.x ? -1 : 1)));
    const y = f.minY + Math.random() * (f.maxY - f.minY);
    const sprite = this.add.sprite(x, y, "boar_idle").setOrigin(0.5, FEET.boar);
    // tap area: the body inside the 92x92 canvas, a bit generous for fingers
    sprite.setInteractive(new Phaser.Geom.Rectangle(4, 12, 84, 62), Phaser.Geom.Rectangle.Contains);
    sprite.play("boar_walk");
    this.boars.push({ id: this.nextBoarId++, sprite, ai: newCharger({ x, y }), hp: H.boar.maxHp, hits: EMPTY_HIT_LOG, dead: false });
  }

  private liveBoar(id: number | null): Boar | undefined {
    return id === null ? undefined : this.boars.find((b) => b.id === id && !b.dead);
  }

  private loop(key: string) {
    if (this.hunter.anims.currentAnim?.key !== key || !this.hunter.anims.isPlaying) this.hunter.play(key);
  }

  private hitStop(ms: number) {
    this.hitStopMs = Math.max(this.hitStopMs, ms);
    this.anims.pauseAll();
  }

  private popText(x: number, y: number, text: string, color: string, ms = 700) {
    const t = this.add
      .text(x, y, text, { fontFamily: "monospace", fontSize: "10px", color, stroke: "#000000", strokeThickness: 3 })
      .setOrigin(0.5)
      .setDepth(900);
    this.tweens.add({ targets: t, y: y - 14, alpha: 0, duration: ms, ease: "Cubic.easeOut", onComplete: () => t.destroy() });
  }

  private syncLayers() {
    const sx = this.cameras.main.scrollX;
    for (const l of this.layers) l.tile.tilePositionX = sx * l.factor;
  }

  private drawHud() {
    const p = this.player;
    this.hud.clear();
    this.hud.fillStyle(0x000000, 0.6).fillRect(6, 16, 82, 8);
    this.hud.fillStyle(0xe04040, 1).fillRect(7, 17, (80 * p.hp) / H.player.maxHp, 6);
  }
}
```
