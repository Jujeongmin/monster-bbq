## Global Constraints

- 재화를 포함한 세이브 전체는 Global User State 하나(`save` 키)에 두고 `$lock(`save:${account}`)` 안에서 한 번에 바꾼다. `$asset`은 쓰지 않는다(설계 7장).

```ts
// The whole save of one account, kept in its Global User State under "save" (spec 7).
export interface SaveData {
  version: 1;
  day: number;
  gold: number;
  meat: { premium: number; normal: number };
}

export function newSave(): SaveData {
  return { version: 1, day: 1, gold: 0, meat: { premium: 0, normal: 0 } };
}

function count(v: unknown, min = 0): number {
  return typeof v === "number" && Number.isFinite(v) ? Math.max(min, Math.floor(v)) : min;
}

// Whatever is stored (or nothing): a usable save. Unknown versions and garbage start over; broken
// fields are repaired, so a bad save never stops the game.
export function readSave(raw: unknown): SaveData {
  const r = raw as Partial<SaveData> | null;
  if (!r || typeof r !== "object" || r.version !== 1) return newSave();
  const meat = (r.meat ?? {}) as Partial<SaveData["meat"]>;
  return { version: 1, day: count(r.day, 1), gold: count(r.gold), meat: { premium: count(meat.premium), normal: count(meat.normal) } };
}
```

```ts
import { BALANCE } from "./balance";
import type { SaveData } from "./save";

type Meat = SaveData["meat"];
export type Action = { type: "huntResult"; meat: Meat } | { type: "sellAll" } | { type: "endDay" };

const isCount = (v: unknown): v is number => typeof v === "number" && Number.isInteger(v) && v >= 0;

// The server's rules for every change to a save (the in-page LocalTransport runs the same code).
// `action` comes from the client, so it is checked here; a rejected action leaves the save as it was.
export function applyAction(save: SaveData, action: unknown): { save: SaveData; error?: string } {
  const a = action as { type?: unknown; meat?: Partial<Meat> | null } | null;
  if (!a || typeof a !== "object") return { save, error: "bad_action" };
  switch (a.type) {
    case "huntResult": {
      const m = a.meat;
      if (!m || !isCount(m.premium) || !isCount(m.normal) || m.premium + m.normal > BALANCE.economy.maxMeatPerHunt) return { save, error: "bad_hunt" };
      return { save: { ...save, meat: { premium: save.meat.premium + m.premium, normal: save.meat.normal + m.normal } } };
    }
    case "sellAll": {
      const p = BALANCE.economy.meatPrice;
      return { save: { ...save, gold: save.gold + save.meat.premium * p.premium + save.meat.normal * p.normal, meat: { premium: 0, normal: 0 } } };
    }
    case "endDay":
      return { save: { ...save, day: save.day + 1 } };
    default:
      return { save, error: "bad_action" };
  }
}

// Knocked out on the hunt: half of each grade is lost (spec 3).
export function koPenalty(meat: Meat): Meat {
  return { premium: Math.floor(meat.premium / 2), normal: Math.floor(meat.normal / 2) };
}
```

```ts
// Online when the build has a Verse8 project id (.env's VITE_AGENT8_VERSE), unless a development
// build was opened with ?local to play against the in-page server instead.
export function wantsOnline(verse: string | undefined, search: string, dev: boolean): boolean {
  if (!verse) return false;
  if (dev && new URLSearchParams(search).has("local")) return false;
```

```ts
import { applyAction, type Action } from "../../shared/actions";
import { readSave, type SaveData } from "../../shared/save";

// How the game reaches "the server": load the save, ask for a change.
export interface Transport {
  load(): Promise<SaveData>;
  act(action: Action): Promise<SaveData>;
}

// The server's rules run in the page against browser storage: local development and tests.
export class LocalTransport implements Transport {
  constructor(
    private readonly storage: Pick<Storage, "getItem" | "setItem">,
    private readonly key = "monster-bbq:save",
  ) {}

  async load(): Promise<SaveData> {
    const raw = this.storage.getItem(this.key);
    return readSave(raw ? JSON.parse(raw) : undefined);
  }

  async act(action: Action): Promise<SaveData> {
    const result = applyAction(await this.load(), action);
    if (result.error) throw new Error(result.error);
    this.storage.setItem(this.key, JSON.stringify(result.save));
    return result.save;
  }
}

export const CALL_TIMEOUT_MS = 15_000;

// The part of @agent8/gameserver's server object this needs (tests stand in for it).
export interface RemoteCaller {
  remoteFunction(fn: string, args?: unknown[]): Promise<unknown>;
}

export function withTimeout<T>(p: Promise<T>, ms: number): Promise<T> {
  return new Promise((resolve, reject) => {
    const id = setTimeout(() => reject(new Error("timeout")), ms);
    p.then(
      (v) => (clearTimeout(id), resolve(v)),
      (e) => (clearTimeout(id), reject(e)),
    );
  });
}

function asSave(reply: unknown): SaveData {
  const r = reply as SaveData | null;
  if (!r || typeof r !== "object" || r.version !== 1) throw new Error("bad_reply");
  return readSave(r);
}

// The real server: server/src/server.ts over Verse8's remoteFunction.
export class Verse8Transport implements Transport {
  constructor(private readonly server: RemoteCaller, private readonly timeoutMs = CALL_TIMEOUT_MS) {}

  async load(): Promise<SaveData> {
    return asSave(await withTimeout(this.server.remoteFunction("load"), this.timeoutMs));
  }

  async act(action: Action): Promise<SaveData> {
    return asSave(await withTimeout(this.server.remoteFunction("act", [action]), this.timeoutMs));
  }
}
```

```ts
import { applyAction } from "../../shared/actions";
import { readSave, type SaveData } from "../../shared/save";

// The whole save lives in the account's Global User State under SAVE_KEY and changes in one write
// under the account's lock (spec 7: no $asset).
const SAVE_KEY = "save";

export class Server {
  async load(): Promise<SaveData> {
    const state = await $global.getMyState();
    return readSave(state?.[SAVE_KEY]);
  }

  // Every change goes through the shared rules; the action is the client's word and is checked there.
  async act(action: unknown): Promise<SaveData> {
    return $lock(`save:${$sender.account}`, async () => {
      const state = await $global.getMyState();
      const result = applyAction(readSave(state?.[SAVE_KEY]), action);
      if (result.error) throw new Error(result.error);
      await $global.updateMyState({ [SAVE_KEY]: result.save });
      return result.save;
    });
  }
}
```

```ts
import { en } from "./en";
import { ko } from "./ko";

export type Locale = "ko" | "en";
export type StringKey = keyof typeof ko;
const DICTS: Record<Locale, Record<StringKey, string>> = { ko, en };
let current: Locale = "ko";

// ?lang=ko|en wins; otherwise Korean browsers get Korean and everyone else English.
export function detectLocale(search: string, navigatorLanguage: string): Locale {
  const asked = new URLSearchParams(search).get("lang");
  if (asked === "ko" || asked === "en") return asked;
  return navigatorLanguage.toLowerCase().startsWith("ko") ? "ko" : "en";
}

export function setLocale(l: Locale) {
  current = l;
}

export function t(key: StringKey, vars: Record<string, string | number> = {}): string {
  return DICTS[current][key].replace(/\{(\w+)\}/g, (_, k: string) => String(vars[k] ?? `{${k}}`));
}
```
